Banco de Dados no Google Drive (Google Sheets)

Conecte este app diretamente a uma Planilha Google privada no seu Google Drive sem precisar pagar servidores.

Status: Não Conectado (Salvando no Navegador)

Passo a Passo (Leva 2 minutos)

  1. 1
    Crie uma Planilha Google no Drive: Acesse sheets.new e dê um nome (ex: Controle Financeiro Pessoal).
  2. 2
    Abra o Apps Script: Na planilha, clique no menu superior em Extensões > Apps Script. Apague qualquer código que estiver lá e cole o código abaixo:
Código.gs (Google Apps Script) 
function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Lancamentos");
  
  // Cria a aba e o cabeçalho automaticamente se não existir
  if (!sheet) {
    sheet = ss.insertSheet("Lancamentos");
    sheet.appendRow(["id", "date", "description", "amount", "type", "debtCategory", "category", "status", "recurring"]);
    sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#4f46e5").setFontColor("#ffffff");
  }

  // Se for POST com dados para salvar/sincronizar
  if (e.postData && e.postData.contents) {
    var payload = JSON.parse(e.postData.contents);
    if (payload.action === "syncAll" && Array.isArray(payload.transactions)) {
      sheet.clearContents();
      sheet.appendRow(["id", "date", "description", "amount", "type", "debtCategory", "category", "status", "recurring"]);
      sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#4f46e5").setFontColor("#ffffff");
      
      var rows = payload.transactions.map(function(t) {
        return [t.id, t.date, t.description, Number(t.amount), t.type, t.debtCategory, t.category, t.status, t.recurring ? "Sim" : "Não"];
      });
      
      if (rows.length > 0) {
        sheet.getRange(2, 1, rows.length, 9).setValues(rows);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // Padrão GET: Lê todas as linhas da planilha
  var data = sheet.getDataRange().getValues();
  var transactions = [];
  for (var i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    var rawDate = data[i][1];
    var formattedDate = rawDate instanceof Date ? Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : String(rawDate);
    transactions.push({
      id: String(data[i][0]),
      date: formattedDate,
      description: String(data[i][2]),
      amount: Number(data[i][3]) || 0,
      type: String(data[i][4]),
      debtCategory: String(data[i][5]),
      category: String(data[i][6]),
      status: String(data[i][7]),
      recurring: data[i][8] === "Sim" || data[i][8] === true
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "success", transactions: transactions }))
    .setMimeType(ContentService.MimeType.JSON);
}
